let state = { db: [...SEED], calc: {}, notes: {} };
let currentFilter = "all";
let searchQuery = "";

const $ = id => document.getElementById(id);
const num = v => { const x = parseFloat(v); return isNaN(x) ? null : x; };
const fmt = (x, d = 0) => (x == null || isNaN(x)) ? "—" : x.toLocaleString("ka-GE", { maximumFractionDigits: d });

function saveState() {
  try { localStorage.setItem("factory_platform_v2", JSON.stringify(state)); } catch(e) {}
}

function loadState() {
  try {
    const saved = localStorage.getItem("factory_platform_v2");
    if (saved) {
      const parsed = JSON.parse(saved);
      if (parsed.db && Array.isArray(parsed.db)) state.db = parsed.db;
      if (parsed.calc) state.calc = parsed.calc;
      if (parsed.notes) state.notes = parsed.notes;
    }
  } catch(e) {}
}

function renderCards() {
  const container = $("cards-container");
  let filtered = currentFilter === "all" ? state.db : state.db.filter(x => x.cat === currentFilter);
  
  if (searchQuery) {
    const q = searchQuery.toLowerCase();
    filtered = filtered.filter(x => 
      (x.title + " " + x.desc + " " + (x.key || "")).toLowerCase().includes(q)
    );
  }

  if (!filtered.length) {
    container.innerHTML = '<p class="mu">ჩანაწერები ვერ მოიძებნა.</p>';
    return;
  }

  container.innerHTML = filtered.map(item => {
    const realIndex = state.db.indexOf(item);
    return `
      <div class="card">
        <button class="x" data-del="${realIndex}" title="წაშლა">×</button>
        <h3>${item.title}</h3>
        ${item.key ? `<span class="k">${item.key}</span>` : ""}
        <p style="margin:8px 0 0">${item.desc}</p>
      </div>
    `;
  }).join("");
}
